Add a second constructor that create a time deposit starting from a model, so as to copy the information in the instrument used. It safer not to have the reference of the model because it could change over time, but the conditions for the single time deposit has to remain the same till the end.

factory TimeDeposit.fromModel(TimeDepositModel model, String name, double initialValue,
                              {DateTime? startingDate, int? id, int? day, int? month, bool includeStart = false, bool includeEnd = false, double? currentValue}){
  /* define missing attributes */
  DateTime firstDate = startingDate ?? DateTime.now();

  /* calculate ending date by adding the model's duration to the starting date */
  DateTime endingDate = DateTime(
    firstDate.year + model.duration.years,
    firstDate.month + model.duration.months,
    firstDate.day,
  ).add(Duration(days: model.duration.days));

  /* for day of month and month year, pickup the
  one of the starting date */
  int dayOfMonth = day ?? firstDate.day;
  int? monthOfYear;
  if(model.payoutFrequency.index >= DurationType.annual.index){
    monthOfYear = month ?? firstDate.month;
  }

  /* create the object */
  return TimeDeposit(
    name,
    model.providerType,
    initialValue,
    model.taxRate,
    model.interestRate,
    firstDate,
    model.payoutFrequency,
    dayOfMonth,
    monthOfYear: monthOfYear,
    includeStart: includeStart,
    includeEnd: includeEnd,
    endingDate: endingDate,
    id: id ?? -1,
    currentValue: currentValue,
    duration: model.duration.clone(),
    provider: model.provider?.clone(),
    instrumentUsed: model.name,
    isCashable: model.isCashable,
    earlyWithdrawalInterestRate: model.earlyWithdrawalInterestRate,
  );
}
