Future<Map<String, dynamic>> exportKeyData()

Function to export the data of the crypto key with salt and all the elements to share this information in public spaces without compromising the confidentiality

Future<Map<String, dynamic>> exportKeyData() async {
  /* first check if the user is authenticated,
  because this operation can be done only for keys
  belonging to accounts */
  if (!isAuthenticated) {
    throw Exception(
      "Exporting key data operation must be done only with a local account",
    );
  }

  /* get the various parameters from the secure storage */
  final storedValue = await _secureStorage.read(
    key: "$_USERKEYPREFIX$_username",
  );
  if (storedValue == null) {
    throw Exception(
      "No result found inside the secure storage of cryptography keys",
    );
  }

  final decodedData = jsonDecode(storedValue);
  if (decodedData is! Map<String, dynamic>) {
    throw Exception("Stored key data is not valid");
  }

  final userKey = decodedData['userKey'];
  final salt = decodedData['salt'];
  final lockerKey = SecretKey(base64Decode(decodedData['lockerKey']));

  /* encrypt the user key and return
  the data as a json format */
  final encryptedUserKey = base64Encode(
    await _encryptDataWithPassword(base64Decode(userKey), lockerKey),
  );

  return {'username': _username, 'salt': salt, 'userKey': encryptedUserKey};
}
