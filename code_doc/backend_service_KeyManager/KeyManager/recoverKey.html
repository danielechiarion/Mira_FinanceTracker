Future<bool> recoverKey()

Function to recover a key from the secure storage and convert it in the right way to be used. Return a boolean value indicating if the key has been found or not

Implementation

Future<bool> recoverKey() async {
  if (_activeKey != null) return true;
  final storageKey = isAuthenticated
      ? "$_USERKEYPREFIX$_username"
      : _APPLICATIONDATABASE;

  String? storedValue;
  try {
    storedValue = await _secureStorage.read(key: storageKey);
  } catch (e) {
    // print('❌ [KeyManager] read($storageKey) failed: $e\n$st');
    rethrow;
  }
  // print('🔑 [KeyManager] $storageKey found=${storedValue != null}');

  /* if no stored value has been found,
  generate a random key if it's for the application */
  if ((storedValue == null || storedValue.isEmpty) && !isAuthenticated) {
    await generateRandomKey();
    return false;
  }
  /* for the user try to get the user key
  otherwise return if still not available */
  if (isAuthenticated) {
    try {
      storedValue = await _secureStorage.read(key: storageKey);
    } catch (e) {
      // print('❌ [KeyManager] read($storageKey) failed: $e\n$st');
      rethrow; // never generate a new key after a reading error
    }
    // print('🔑 [KeyManager] $storageKey found=${storedValue != null}');

    final decodedData = jsonDecode(storedValue!);
    if (decodedData is! Map<String, dynamic>) {
      throw Exception("Stored key data is not valid");
    }

    final userKeyBase64 = decodedData['userKey'] as String?;
    if (userKeyBase64 == null || userKeyBase64.isEmpty) {
      throw Exception("Missing userKey in stored data");
    }

    _activeKey = SecretKey(base64Decode(userKeyBase64));
  } else {
    _activeKey = SecretKey(base64Decode(storedValue!));
  }

  return true;
}
